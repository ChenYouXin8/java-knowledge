## 📈 学习统计总览

```dataviewjs
const allFiles = dv.pages('""').where(p => p.file.path !== this.file.path && p.file.folder);

// 总笔记数
const total = allFiles.length;

// 有任务的笔记
const withTasks = allFiles.where(p => p.file.tasks);
const completed = withTasks.where(p => p.file.tasks.every(t => t.completed)).length;
const inProgress = withTasks.where(p => p.file.tasks.some(t => !t.completed)).length;
const noTask = total - withTasks.length;

// 各分类
const folders = [...new Set(allFiles.file.folder)].filter(Boolean).sort();

dv.header(2, "📈 统计总览");
dv.table(
  ["指标", "数量"],
  [
    ["📚 笔记总数", total],
    ["✅ 已完成", completed],
    ["🟨 进行中", inProgress],
    ["⬜ 暂无任务", noTask],
    ["📂 分类数", folders.length]
  ]
);
```
